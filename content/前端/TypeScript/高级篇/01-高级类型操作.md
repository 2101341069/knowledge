# 高级类型操作

## 一、typeof 类型查询

### 1.1 获取值的类型

```typescript
const user = {
  name: 'Tom',
  age: 25
}

// 获取 user 变量的类型
type User = typeof user
// { name: string; age: number }

// 使用
const anotherUser: User = {
  name: 'Jerry',
  age: 30
}
```

### 1.2 与 ReturnType 结合

```typescript
function createPerson() {
  return { name: 'Tom', age: 25 }
}

// 先 typeof 获取函数类型，再 ReturnType 取返回值
type Person = ReturnType<typeof createPerson>
// { name: string; age: number }
```

## 二、keyof 索引类型查询

### 2.1 获取对象键的联合类型

```typescript
interface User {
  id: number
  name: string
  email: string
}

// 获取所有键的联合类型
type UserKeys = keyof User
// 'id' | 'name' | 'email'
```

### 2.2 类型安全的属性访问

```typescript
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key]
}

const user: User = { id: 1, name: 'Tom', email: 'tom@example.com' }

const id = getProperty(user, 'id')      // ✅ number
const name = getProperty(user, 'name')  // ✅ string
const wrong = getProperty(user, 'age')  // ❌ 'age' 不存在
```

## 三、索引访问类型（Indexed Access Types）

### 3.1 通过索引获取属性类型

```typescript
interface User {
  id: number
  name: string
  address: {
    city: string
    street: string
  }
}

// 获取单个属性类型
type UserId = User['id']  // number

// 获取嵌套属性类型
type City = User['address']['city']  // string

// 联合类型索引，获取多个属性类型的联合
type UserFields = User['id' | 'name']  // number | string

// keyof 获取所有
type AllUserFields = User[keyof User]  // number | string | { city: string; street: string }
```

## 四、条件类型（Conditional Types）

### 4.1 基础语法

```typescript
// 语法：A extends B ? X : Y
type IsString<T> = T extends string ? 'yes' : 'no'

type A = IsString<'hello'>  // 'yes'
type B = IsString<42>       // 'no'
```

### 4.2 集合操作

```typescript
// 从 T 中排除可以赋值给 U 的类型（内置 Exclude）
type MyExclude<T, U> = T extends U ? never : T

type Result = MyExclude<'a' | 'b' | 'c', 'a'>  // 'b' | 'c'

// 从 T 中提取可以赋值给 U 的类型（内置 Extract）
type MyExtract<T, U> = T extends U ? T : never

type Result2 = MyExtract<'a' | 'b' | 'c', 'a' | 'b'>  // 'a' | 'b'
```

### 4.3 分布式条件类型

当 T 是联合类型时，会**分布式**执行：

```typescript
// T = A | B | C
// T extends U ? X : Y
// 等价于
// (A extends U ? X : Y) | (B extends U ? X : Y) | (C extends U ? X : Y)

type StringOrNumber<T> = T extends string ? 'string' : 'number'

type Test = StringOrNumber<'a' | 1 | 'b' | 2>
// 'string' | 'number' | 'string' | 'number'
// 最终 'string' | 'number'
```

## 五、infer 类型推断

### 5.1 基础用法

```typescript
// 获取函数返回值类型（内置 ReturnType）
type MyReturnType<T> = T extends (...args: any[]) => infer R ? R : never

function add(a: number, b: number): number {
  return a + b
}

type AddResult = MyReturnType<typeof add>  // number
```

### 5.2 获取函数参数类型

```typescript
// 获取函数第一个参数类型
type FirstParam<T> = T extends (first: infer P, ...rest: any[]) => any ? P : never

type First = FirstParam<typeof add>  // number
```

### 5.3 提取 Promise 的类型

```typescript
// 提取 Promise 包裹的类型
type UnwrapPromise<T> = T extends Promise<infer U> ? U : T

type A = UnwrapPromise<Promise<string>>  // string
type B = UnwrapPromise<number>            // number
```

### 5.4 提取数组元素类型

```typescript
type ArrayElement<T> = T extends (infer E)[] ? E : never

type Num = ArrayElement<number[]>  // number
type Str = ArrayElement<string[]>  // string
```

## 六、类型收窄（Narrowing）

### 6.1 typeof 收窄

```typescript
function padLeft(padding: number | string, input: string) {
  if (typeof padding === 'number') {
    // 这里 padding 是 number
    return ' '.repeat(padding) + input
  }
  // 这里 padding 是 string
  return padding + input
}
```

### 6.2 真值收窄

```typescript
function printAll(strs: string | string[] | null) {
  if (strs) {
    if (typeof strs === 'object') {
      // string[]
      strs.forEach(s => console.log(s))
    } else {
      // string
      console.log(strs)
    }
  }
}
```

### 6.3 相等性收窄

```typescript
function example(x: string | number, y: string | boolean) {
  if (x === y) {
    // x 和 y 都是 string
    x.toUpperCase()
    y.toLowerCase()
  }
}
```

### 6.4 in 操作符收窄

```typescript
interface Circle {
  kind: 'circle'
  radius: number
}

interface Square {
  kind: 'square'
  side: number
}

type Shape = Circle | Square

function getArea(shape: Shape) {
  if ('radius' in shape) {
    // Circle
    return Math.PI * shape.radius ** 2
  }
  // Square
  return shape.side ** 2
}
```

### 6.5 可辨识联合（Discriminated Unions）

```typescript
interface Circle {
  kind: 'circle'  // 可辨识的 tag
  radius: number
}

interface Square {
  kind: 'square'  // 可辨识的 tag
  side: number
}

type Shape = Circle | Square

function getArea(shape: Shape) {
  switch (shape.kind) {
    case 'circle':
      return Math.PI * shape.radius ** 2
    case 'square':
      return shape.side ** 2
  }
}
```

### 6.6 类型谓词（Type Predicates）

```typescript
interface Fish {
  swim(): void
}

interface Bird {
  fly(): void
}

// is 关键字
function isFish(pet: Fish | Bird): pet is Fish {
  return (pet as Fish).swim !== undefined
}

function move(pet: Fish | Bird) {
  if (isFish(pet)) {
    pet.swim()  // ✅ Fish
  } else {
    pet.fly()   // ✅ Bird
  }
}
```
