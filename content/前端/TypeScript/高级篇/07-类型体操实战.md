# 类型体操实战

"类型体操"指利用 TypeScript 类型系统的各种特性，实现复杂的类型推导和变换。本章通过经典题目带你深入理解类型系统。

---

## 一、基础工具类型实现

### 1.1 实现 MyPartial

```typescript
// 将所有属性变为可选
type MyPartial<T> = {
  [P in keyof T]?: T[P]
}

// 测试
interface User {
  id: number
  name: string
  email: string
}

type PartialUser = MyPartial<User>
// { id?: number; name?: string; email?: string }
```

---

### 1.2 实现 MyRequired

```typescript
// 将所有属性变为必选
type MyRequired<T> = {
  [P in keyof T]-?: T[P]
}

type RequiredUser = MyRequired<PartialUser>
// { id: number; name: string; email: string }
```

---

### 1.3 实现 MyReadonly

```typescript
type MyReadonly<T> = {
  readonly [P in keyof T]: T[P]
}
```

---

### 1.4 实现 MyPick

```typescript
type MyPick<T, K extends keyof T> = {
  [P in K]: T[P]
}

type UserPreview = MyPick<User, 'id' | 'name'>
// { id: number; name: string }
```

---

### 1.5 实现 MyOmit

```typescript
// 方法一：结合 Exclude
type MyOmit<T, K extends keyof any> = MyPick<T, Exclude<keyof T, K>>

// 方法二：键名重映射
type MyOmit2<T, K extends keyof any> = {
  [P in keyof T as P extends K ? never : P]: T[P]
}

type SafeUser = MyOmit<User, 'email'>
// { id: number; name: string }
```

---

### 1.6 实现 MyRecord

```typescript
type MyRecord<K extends keyof any, V> = {
  [P in K]: V
}

type StringRecord = MyRecord<'a' | 'b' | 'c', number>
// { a: number; b: number; c: number }
```

## 二、条件类型进阶

### 2.1 实现 MyExclude

```typescript
type MyExclude<T, U> = T extends U ? never : T

type Result = MyExclude<'a' | 'b' | 'c', 'a' | 'b'>  // 'c'
```

---

### 2.2 实现 MyExtract

```typescript
type MyExtract<T, U> = T extends U ? T : never

type Result = MyExtract<'a' | 'b' | 'c', 'a' | 'd'>  // 'a'
```

---

### 2.3 实现 MyReturnType

```typescript
type MyReturnType<T extends (...args: any) => any> =
  T extends (...args: any) => infer R ? R : never

function greet(name: string): string {
  return `Hello, ${name}!`
}

type GreetReturn = MyReturnType<typeof greet>  // string
```

---

### 2.4 实现 MyParameters

```typescript
type MyParameters<T extends (...args: any) => any> =
  T extends (...args: infer P) => any ? P : never

type GreetParams = MyParameters<typeof greet>  // [string]
```

---

### 2.5 实现 MyAwaitedType（递归展开 Promise）

```typescript
type MyAwaited<T> =
  T extends null | undefined ? T :
  T extends object & { then(onfulfilled: infer F, ...args: any): any }
    ? F extends (value: infer V, ...args: any) => any
      ? MyAwaited<V>
      : never
    : T

type A = MyAwaited<Promise<string>>           // string
type B = MyAwaited<Promise<Promise<number>>>  // number
type C = MyAwaited<number>                    // number
```

## 三、递归类型

### 3.1 深层 Readonly

```typescript
type DeepReadonly<T> = {
  readonly [P in keyof T]:
    T[P] extends object
      ? T[P] extends Function
        ? T[P]
        : DeepReadonly<T[P]>
      : T[P]
}

interface Config {
  server: {
    host: string
    port: number
    options: {
      timeout: number
    }
  }
}

type ReadonlyConfig = DeepReadonly<Config>
// 所有嵌套层级都变为 readonly
```

---

### 3.2 深层 Partial

```typescript
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object
    ? T[P] extends Function
      ? T[P]
      : DeepPartial<T[P]>
    : T[P]
}

type PartialConfig = DeepPartial<Config>
// { server?: { host?: string; port?: number; options?: { timeout?: number } } }
```

---

### 3.3 Flatten 展平数组类型

```typescript
type Flatten<T extends any[]> =
  T extends [infer First, ...infer Rest]
    ? First extends any[]
      ? [...Flatten<First>, ...Flatten<Rest>]
      : [First, ...Flatten<Rest>]
    : []

type Flat = Flatten<[1, [2, 3], [4, [5, 6]]]>
// [1, 2, 3, 4, 5, 6]
```

## 四、字符串类型操作

### 4.1 字符串转联合类型

```typescript
// 'a,b,c' -> 'a' | 'b' | 'c'
type Split<S extends string, D extends string> =
  S extends `${infer H}${D}${infer T}` ? H | Split<T, D> : S

type Parts = Split<'a,b,c', ','>  // 'a' | 'b' | 'c'
```

---

### 4.2 驼峰转下划线

```typescript
type CamelToSnake<S extends string> =
  S extends `${infer H}${infer T}`
    ? H extends Uppercase<H>
      ? `_${Lowercase<H>}${CamelToSnake<T>}`
      : `${H}${CamelToSnake<T>}`
    : S

type Snake = CamelToSnake<'helloWorld'>         // 'hello_world'
type Snake2 = CamelToSnake<'getUserProfile'>    // 'get_user_profile'
```

---

### 4.3 下划线转驼峰

```typescript
type SnakeToCamel<S extends string> =
  S extends `${infer H}_${infer T}`
    ? `${H}${Capitalize<SnakeToCamel<T>>}`
    : S

type Camel = SnakeToCamel<'hello_world'>         // 'helloWorld'
type Camel2 = SnakeToCamel<'get_user_profile'>   // 'getUserProfile'
```

## 五、元组与数组操作

### 5.1 元组头尾操作

```typescript
// 获取第一个元素类型
type Head<T extends any[]> = T extends [infer H, ...any[]] ? H : never
type H = Head<[1, 2, 3]>  // 1

// 获取最后一个元素类型
type Last<T extends any[]> = T extends [...any[], infer L] ? L : never
type L = Last<[1, 2, 3]>  // 3

// 去掉第一个元素
type Tail<T extends any[]> = T extends [any, ...infer R] ? R : never
type T = Tail<[1, 2, 3]>  // [2, 3]

// 去掉最后一个元素
type Init<T extends any[]> = T extends [...infer I, any] ? I : never
type I = Init<[1, 2, 3]>  // [1, 2]
```

---

### 5.2 元组翻转

```typescript
type Reverse<T extends any[]> =
  T extends [infer H, ...infer T]
    ? [...Reverse<T>, H]
    : []

type Rev = Reverse<[1, 2, 3]>  // [3, 2, 1]
```

---

### 5.3 获取元组长度

```typescript
type Length<T extends any[]> = T['length']

type Len = Length<[1, 2, 3]>  // 3

// 利用元组长度实现数字类型加法
type BuildTuple<N extends number, T extends any[] = []> =
  T['length'] extends N ? T : BuildTuple<N, [...T, any]>

type Add<A extends number, B extends number> =
  [...BuildTuple<A>, ...BuildTuple<B>]['length']

type Sum = Add<3, 4>  // 7
```

## 六、对象类型高级操作

### 6.1 获取可选属性键

```typescript
type OptionalKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? K : never
}[keyof T]

interface Foo {
  required: string
  optional?: number
  maybeUndefined: string | undefined
}

type Optional = OptionalKeys<Foo>  // 'optional'
```

---

### 6.2 获取必选属性键

```typescript
type RequiredKeys<T> = {
  [K in keyof T]-?: {} extends Pick<T, K> ? never : K
}[keyof T]

type Required = RequiredKeys<Foo>  // 'required' | 'maybeUndefined'
```

---

### 6.3 合并两个对象类型

```typescript
// 合并两个类型，B 的属性覆盖 A 的同名属性
type Merge<A, B> = Omit<A, keyof B> & B

interface A {
  id: number
  name: string
  age: number
}

interface B {
  name: string  // 覆盖 A.name
  email: string // 新增
}

type Merged = Merge<A, B>
// { id: number; age: number; name: string; email: string }
```

---

### 6.4 对象类型的 XOR（互斥属性）

```typescript
// 两个类型的异或：要么有 A 的属性，要么有 B 的属性，但不能两者都有
type Without<T, U> = { [P in Exclude<keyof T, keyof U>]?: never }

type XOR<T, U> = (T | U) extends object
  ? (Without<T, U> & U) | (Without<U, T> & T)
  : T | U

type Props = XOR<
  { type: 'a'; aOnly: string },
  { type: 'b'; bOnly: number }
>

const a: Props = { type: 'a', aOnly: 'hello' }    // ✅
const b: Props = { type: 'b', bOnly: 42 }          // ✅
// const c: Props = { type: 'a', bOnly: 42 }        // ❌
```

## 七、经典挑战题

### 7.1 实现 IsNever

```typescript
type IsNever<T> = [T] extends [never] ? true : false

type A = IsNever<never>   // true
type B = IsNever<string>  // false
type C = IsNever<any>     // false
```

### 7.2 实现 IsAny

```typescript
type IsAny<T> = 0 extends (1 & T) ? true : false

type A = IsAny<any>     // true
type B = IsAny<string>  // false
type C = IsAny<never>   // false
```

### 7.3 实现 IsUnion

```typescript
type IsUnion<T, C extends T = T> = (
  T extends any ? (C extends T ? true : unknown) : never
) extends true ? false : true

type A = IsUnion<string | number>  // true
type B = IsUnion<string>           // false
type C = IsUnion<never>            // false
```

### 7.4 实现 TupleToUnion

```typescript
type TupleToUnion<T extends any[]> = T[number]

type Union = TupleToUnion<[1, 2, 3]>  // 1 | 2 | 3
type StrUnion = TupleToUnion<['a', 'b', 'c']>  // 'a' | 'b' | 'c'
```

### 7.5 实现 UnionToTuple（困难）

```typescript
// 利用函数重载将联合转元组
type UnionToIntersection<U> = (U extends any ? (k: U) => void : never) extends
  (k: infer I) => void ? I : never

type LastUnion<U> = UnionToIntersection<U extends any ? () => U : never> extends
  () => infer R ? R : never

type UnionToTuple<U, T extends any[] = []> =
  [U] extends [never]
    ? T
    : UnionToTuple<Exclude<U, LastUnion<U>>, [LastUnion<U>, ...T]>

type Tuple = UnionToTuple<'a' | 'b' | 'c'>  // ['a', 'b', 'c']
```

## 八、实战应用

### 8.1 类型安全的路由参数

```typescript
// 提取路由中的参数名
type ExtractParams<T extends string> =
  T extends `${infer _}:${infer Param}/${infer Rest}`
    ? Param | ExtractParams<`/${Rest}`>
    : T extends `${infer _}:${infer Param}`
    ? Param
    : never

// 生成参数对象类型
type RouteParams<T extends string> = {
  [K in ExtractParams<T>]: string
}

type UserRoute = RouteParams<'/users/:id'>
// { id: string }

type PostRoute = RouteParams<'/users/:userId/posts/:postId'>
// { userId: string; postId: string }

function navigate<T extends string>(path: T, params: RouteParams<T>): string {
  let result = path
  Object.entries(params).forEach(([key, value]) => {
    result = result.replace(`:${key}`, value as string)
  })
  return result
}

navigate('/users/:id', { id: '123' })
navigate('/users/:userId/posts/:postId', { userId: '1', postId: '2' })
```

## 总结

类型体操的核心工具：

| 工具 | 用途 |
|------|------|
| `keyof T` | 获取对象所有键的联合 |
| `T[K]` | 获取索引类型（属性类型） |
| `[P in keyof T]` | 映射类型，遍历属性 |
| `T extends U ? X : Y` | 条件类型 |
| `infer R` | 类型推断，提取类型片段 |
| `[T] extends [never]` | 安全检查 never（避免分布式） |
| 递归类型 | 处理嵌套/任意深度的类型 |

类型体操是掌握 TypeScript 类型系统精髓的最佳方式，推荐挑战 [TypeScript Challenges](https://github.com/type-challenges/type-challenges)
