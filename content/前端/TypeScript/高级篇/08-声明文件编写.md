# 声明文件编写（Declaration Files）

## 一、什么是声明文件？

声明文件（`.d.ts`）用于为**不含类型信息的 JavaScript 代码**提供类型描述。它只包含类型声明，不含任何实现代码。

```
npm 包
  ├── index.js         ← 实现代码（JS）
  └── index.d.ts       ← 类型声明（TS）
```

---

## 二、为什么需要声明文件？

```typescript
// ❌ 没有声明文件，使用 JS 库时没有类型提示
import moment from 'moment'
moment().format()  // moment: any，没有类型检查

// ✅ 有 @types/moment，使用时有完整类型
import moment from 'moment'
moment().format('YYYY-MM-DD')  // 完整的类型提示
```

---

## 三、获取第三方库的声明文件

### 3.1 库自带声明文件

现代 TypeScript 库通常自带 `.d.ts`：

```bash
# 库目录结构
node_modules/axios/
  ├── index.js
  └── index.d.ts   ← 自带
```

---

### 3.2 DefinitelyTyped（@types）

许多 JS 库的类型由社区维护：

```bash
# 安装类型声明
npm install -D @types/lodash
npm install -D @types/node
npm install -D @types/express

# package.json 中 devDependencies
{
  "@types/lodash": "^4.14.0",
  "@types/node": "^20.0.0"
}
```

---

### 3.3 查找策略

TypeScript 按以下顺序查找类型：

1. `node_modules/library/package.json` 的 `types` 或 `typings` 字段
2. `node_modules/library/index.d.ts`
3. `node_modules/@types/library/index.d.ts`
4. `tsconfig.json` 的 `typeRoots` 配置

## 四、编写声明文件

### 4.1 全局变量声明

```typescript
// global.d.ts
// 声明全局变量
declare const __DEV__: boolean
declare const __VERSION__: string
declare const API_URL: string

// 声明全局函数
declare function trackEvent(name: string, data?: Record<string, unknown>): void
declare function requireModule(path: string): any

// 声明全局类
declare class EventBus {
  on(event: string, handler: Function): void
  off(event: string, handler?: Function): void
  emit(event: string, ...args: any[]): void
}

// 扩展 Window 对象
interface Window {
  __APP_CONFIG__: {
    apiUrl: string
    version: string
    env: string
  }
  gtag: (command: string, ...args: any[]) => void
}
```

---

### 4.2 模块声明

```typescript
// 为没有类型的 JS 模块声明类型
declare module 'some-legacy-library' {
  export interface Options {
    key: string
    value: any
    timeout?: number
  }

  export function init(options: Options): void
  export function destroy(): void
  export const version: string

  export default class LegacyClass {
    constructor(options: Options)
    start(): void
    stop(): void
  }
}
```

---

### 4.3 为非 JS 文件声明类型

```typescript
// 声明 CSS 模块
declare module '*.module.css' {
  const classes: Record<string, string>
  export default classes
}

declare module '*.module.scss' {
  const classes: Record<string, string>
  export default classes
}

// 声明图片资源
declare module '*.png' {
  const src: string
  export default src
}

declare module '*.svg' {
  import React from 'react'
  const ReactComponent: React.FunctionComponent<React.SVGProps<SVGSVGElement>>
  export { ReactComponent }
  const src: string
  export default src
}

// 声明 JSON 文件（tsconfig 中 resolveJsonModule: true 时不需要）
declare module '*.json' {
  const value: any
  export default value
}
```

## 五、编写完整的库声明文件

以一个假想的 `my-utils` 库为例：

```typescript
// my-utils/index.d.ts

// ===================== 类型定义 =====================

export interface FormatOptions {
  locale?: string
  currency?: string
  decimals?: number
}

export interface DateFormatOptions {
  format?: 'short' | 'medium' | 'long' | 'iso'
  timezone?: string
}

export type ValidatorFn<T> = (value: T) => boolean | string

// ===================== 函数声明 =====================

/**
 * 格式化数字为货币字符串
 * @param value 数字值
 * @param options 格式化选项
 * @returns 格式化后的字符串
 */
export function formatCurrency(value: number, options?: FormatOptions): string

/**
 * 格式化日期
 */
export function formatDate(date: Date | string | number, options?: DateFormatOptions): string

/**
 * 深拷贝
 */
export function deepClone<T>(value: T): T

/**
 * 防抖函数
 */
export function debounce<T extends (...args: any[]) => any>(
  fn: T,
  delay: number
): (...args: Parameters<T>) => void

/**
 * 节流函数
 */
export function throttle<T extends (...args: any[]) => any>(
  fn: T,
  delay: number
): (...args: Parameters<T>) => void

// ===================== 类声明 =====================

export class Validator<T> {
  constructor(rules: Record<keyof T, ValidatorFn<T[keyof T]>[]>)
  validate(data: T): Record<string, string[]>
  isValid(data: T): boolean
}

// ===================== 命名空间声明 =====================

export namespace Utils {
  function isEmpty(value: any): boolean
  function isNil(value: any): value is null | undefined
  function chunk<T>(arr: T[], size: number): T[][]
}

// ===================== 默认导出 =====================

declare const _default: {
  formatCurrency: typeof formatCurrency
  formatDate: typeof formatDate
  deepClone: typeof deepClone
  debounce: typeof debounce
  throttle: typeof throttle
  Validator: typeof Validator
  Utils: typeof Utils
}

export default _default
```

## 六、模块扩充（Augmentation）

### 6.1 扩展第三方模块

```typescript
// express-augmentation.d.ts
import 'express-serve-static-core'

declare module 'express-serve-static-core' {
  interface Request {
    // 添加自定义属性
    user?: {
      id: string
      email: string
      roles: string[]
    }
    requestId?: string
    startTime?: number
  }

  interface Response {
    // 添加自定义方法
    sendSuccess<T>(data: T, message?: string): void
    sendError(message: string, code?: number): void
  }
}
```

```typescript
// 给 Vue 实例添加全局属性
// vue-augmentation.d.ts
import { ComponentCustomProperties } from 'vue'

declare module '@vue/runtime-core' {
  interface ComponentCustomProperties {
    $formatDate: (date: Date) => string
    $t: (key: string, params?: Record<string, string>) => string
    $http: typeof axios
  }
}
```

---

### 6.2 全局扩充

```typescript
// global-augmentation.d.ts
declare global {
  interface Array<T> {
    // 给数组添加自定义方法（不推荐，容易造成命名冲突）
    first(): T | undefined
    last(): T | undefined
    chunk(size: number): T[][]
  }

  interface String {
    toSnakeCase(): string
    toCamelCase(): string
  }
}

export {}  // 确保文件被视为模块
```

## 七、声明文件的最佳实践

### 7.1 项目结构

```
src/
├── types/
│   ├── global.d.ts         # 全局类型扩充
│   ├── env.d.ts            # 环境变量类型
│   ├── assets.d.ts         # 静态资源类型
│   └── vendors/            # 第三方库的补充声明
│       └── some-lib.d.ts
└── ...

// tsconfig.json
{
  "include": [
    "src/**/*.ts",
    "src/**/*.d.ts"
  ]
}
```

---

### 7.2 tsconfig.json 中的声明文件配置

```json
{
  "compilerOptions": {
    // 生成声明文件
    "declaration": true,
    // 生成声明文件的 sourcemap
    "declarationMap": true,
    // 声明文件输出目录
    "declarationDir": "./types",
    // 自定义类型根目录
    "typeRoots": ["./src/types", "node_modules/@types"],
    // 只引入指定的 @types 包
    "types": ["node", "jest"]
  }
}
```

---

### 7.3 发布包时的声明文件配置

```json
// package.json
{
  "name": "my-library",
  "version": "1.0.0",
  "main": "dist/index.js",
  "types": "dist/index.d.ts",     // 声明文件入口
  "typings": "dist/index.d.ts",   // 同上，另一种写法
  "exports": {
    ".": {
      "types": "./dist/index.d.ts",
      "import": "./dist/index.esm.js",
      "require": "./dist/index.cjs.js"
    }
  },
  "files": [
    "dist"
  ]
}
```

## 八、declare 关键字详解

```typescript
// declare 用于告诉 TS "这个东西在运行时存在，但不是在这里定义的"

// 变量
declare const PI: number
declare let count: number
declare var global: NodeJS.Global

// 函数
declare function require(module: string): any

// 类
declare class Buffer {
  static from(data: string, encoding?: string): Buffer
  toString(encoding?: string): string
}

// 枚举（外部枚举）
declare enum HttpStatus {
  OK = 200,
  NotFound = 404
}

// 命名空间（全局库）
declare namespace JQuery {
  function ajax(url: string, settings?: object): void
  function get(url: string): void
}

declare const $: JQuery
declare const jQuery: JQuery

// 模块
declare module 'path' {
  export function join(...paths: string[]): string
  export function resolve(...paths: string[]): string
}
```

## 总结

- **`.d.ts` 文件**：纯类型声明，不含实现，编译后不生成 JS
- **`declare` 关键字**：告诉编译器某个东西在运行时存在
- **模块扩充**：为第三方库添加/扩充类型
- **全局扩充**：修改 Window、Array 等全局类型
- **发布库**：设置 `package.json` 的 `types` 字段指向 `.d.ts` 文件
