# 函数类型进阶

## 一、回顾：函数基础类型

```typescript
// 参数类型 + 返回值类型
function add(a: number, b: number): number {
  return a + b
}

// 箭头函数
const greet = (name: string): string => `Hello, ${name}!`

// void 返回值
function log(msg: string): void {
  console.log(msg)
}
```

## 二、函数类型定义方式

### 2.1 类型别名定义函数类型

```typescript
// 用 type 定义函数签名
type Add = (a: number, b: number) => number
type Callback<T> = (error: Error | null, result: T) => void
type Predicate<T> = (value: T) => boolean
type Transformer<T, R> = (value: T) => R

// 使用
const add: Add = (a, b) => a + b

const isPositive: Predicate<number> = (n) => n > 0
const toString: Transformer<number, string> = (n) => String(n)
```

---

### 2.2 接口定义函数类型

```typescript
// 用 interface 定义调用签名
interface Formatter {
  (value: string): string
}

interface EventListener {
  (event: MouseEvent): void
}

const upperCase: Formatter = (s) => s.toUpperCase()
```

---

### 2.3 内联函数类型

```typescript
// 参数直接使用函数类型
function applyTwice(fn: (x: number) => number, value: number): number {
  return fn(fn(value))
}

applyTwice(x => x * 2, 3)  // 12
```

## 三、函数重载（Function Overloads）

### 3.1 为什么需要重载？

```typescript
// ❌ 不用重载：返回类型不精确
function process(input: string | number): string | number {
  if (typeof input === 'string') {
    return input.toUpperCase()
  }
  return input * 2
}

const result = process('hello')  // 类型是 string | number，不够精确
```

---

### 3.2 函数重载语法

```typescript
// 先写重载签名（只声明，不实现）
function process(input: string): string
function process(input: number): number

// 再写实现签名（兼容所有重载）
function process(input: string | number): string | number {
  if (typeof input === 'string') {
    return input.toUpperCase()
  }
  return input * 2
}

// 调用时类型精准
const result1 = process('hello')  // string ✅
const result2 = process(42)        // number ✅
```

---

### 3.3 更复杂的重载示例

```typescript
// 根据参数不同返回不同类型
function createElement(tag: 'a'): HTMLAnchorElement
function createElement(tag: 'canvas'): HTMLCanvasElement
function createElement(tag: 'table'): HTMLTableElement
function createElement(tag: string): HTMLElement {
  return document.createElement(tag)
}

// 调用时自动推断返回类型
const anchor = createElement('a')    // HTMLAnchorElement
const canvas = createElement('canvas')  // HTMLCanvasElement
const table = createElement('table')    // HTMLTableElement
```

---

### 3.4 方法重载

```typescript
class DateHelper {
  // 重载签名
  format(date: Date): string
  format(date: string): string
  format(date: number): string
  
  // 实现签名
  format(date: Date | string | number): string {
    const d = new Date(date)
    return d.toLocaleDateString()
  }
}
```

## 四、this 类型

### 4.1 this 参数

TypeScript 允许在函数参数位置声明 `this` 的类型（编译后会被移除）：

```typescript
interface User {
  name: string
  greet(this: User): string  // this 必须是 User 类型
}

const user: User = {
  name: 'Tom',
  greet() {
    return `Hello, ${this.name}`
  }
}

user.greet()  // ✅

// ❌ 脱离对象调用，this 类型不对
const greetFn = user.greet
greetFn()  // ❌ 报错
```

---

### 4.2 回调函数中的 this

```typescript
class EventEmitter {
  private handlers: ((this: EventEmitter, data: any) => void)[] = []

  on(handler: (this: EventEmitter, data: any) => void) {
    this.handlers.push(handler)
  }

  emit(data: any) {
    this.handlers.forEach(h => h.call(this, data))
  }
}
```

---

### 4.3 this 类型与链式调用

```typescript
class Builder {
  private config: Record<string, any> = {}

  set(key: string, value: any): this {
    this.config[key] = value
    return this  // 返回 this 实现链式调用
  }

  build() {
    return this.config
  }
}

class AdvancedBuilder extends Builder {
  setMultiple(items: Record<string, any>): this {
    Object.assign(this.config, items)
    return this
  }
}

// 子类也能正确链式
const result = new AdvancedBuilder()
  .set('a', 1)
  .setMultiple({ b: 2, c: 3 })
  .build()
```

## 五、剩余参数与展开

### 5.1 剩余参数

```typescript
// 剩余参数必须是最后一个
function sum(...numbers: number[]): number {
  return numbers.reduce((a, b) => a + b, 0)
}

sum(1, 2, 3)      // 6
sum(1, 2, 3, 4)   // 10

// 混合参数
function log(prefix: string, ...messages: string[]): void {
  messages.forEach(msg => console.log(`[${prefix}] ${msg}`))
}

log('INFO', 'Server started', 'Port: 3000')
```

---

### 5.2 元组类型与剩余参数

```typescript
// 元组用作剩余参数
function apply<T extends any[], R>(
  fn: (...args: T) => R,
  args: T
): R {
  return fn(...args)
}

apply(Math.max, [1, 2, 3])  // 3
apply(parseInt, ['42', 10]) // 42
```

## 六、可选参数与默认参数

### 6.1 可选参数

```typescript
// 可选参数必须在必选参数后面
function createUser(
  name: string,
  age: number,
  email?: string,    // 可选
  role?: 'admin' | 'user'  // 可选
): User {
  return {
    name,
    age,
    email: email ?? '',
    role: role ?? 'user'
  }
}

createUser('Tom', 25)
createUser('Tom', 25, 'tom@example.com')
createUser('Tom', 25, 'tom@example.com', 'admin')
```

---

### 6.2 默认参数

```typescript
// 默认参数 = 隐式可选
function createConfig(
  host: string = 'localhost',
  port: number = 3000,
  debug: boolean = false
) {
  return { host, port, debug }
}

createConfig()                    // 全用默认值
createConfig('example.com')      // 只改 host
createConfig('example.com', 443) // 改 host 和 port

// 默认参数可以是表达式
function delay(ms: number = Date.now() % 1000) {
  return new Promise(resolve => setTimeout(resolve, ms))
}
```

## 七、函数作为值

### 7.1 高阶函数

```typescript
// 接收函数作为参数
function filter<T>(arr: T[], predicate: (item: T) => boolean): T[] {
  return arr.filter(predicate)
}

// 返回函数
function createMultiplier(factor: number): (n: number) => number {
  return (n) => n * factor
}

const double = createMultiplier(2)
const triple = createMultiplier(3)

double(5)  // 10
triple(5)  // 15
```

---

### 7.2 函数柯里化

```typescript
// 柯里化
function curry<A, B, C>(fn: (a: A, b: B) => C): (a: A) => (b: B) => C {
  return (a) => (b) => fn(a, b)
}

const add = (a: number, b: number) => a + b
const curriedAdd = curry(add)
const add5 = curriedAdd(5)  // (b: number) => number

add5(3)  // 8
add5(10) // 15
```

---

### 7.3 函数组合

```typescript
// 函数组合
function compose<A, B, C>(
  f: (b: B) => C,
  g: (a: A) => B
): (a: A) => C {
  return (a) => f(g(a))
}

const trim = (s: string) => s.trim()
const upper = (s: string) => s.toUpperCase()
const trimAndUpper = compose(upper, trim)

trimAndUpper('  hello world  ')  // 'HELLO WORLD'
```

## 八、异步函数类型

### 8.1 async/await 函数类型

```typescript
// 异步函数返回 Promise<T>
async function fetchUser(id: number): Promise<User> {
  const res = await fetch(`/api/users/${id}`)
  return res.json()
}

// 箭头函数异步
const fetchUsers = async (): Promise<User[]> => {
  const res = await fetch('/api/users')
  return res.json()
}
```

---

### 8.2 错误处理

```typescript
type Result<T, E = Error> = 
  | { success: true; data: T }
  | { success: false; error: E }

async function safeRequest<T>(url: string): Promise<Result<T>> {
  try {
    const res = await fetch(url)
    if (!res.ok) {
      return { success: false, error: new Error(`HTTP ${res.status}`) }
    }
    const data: T = await res.json()
    return { success: true, data }
  } catch (e) {
    return { success: false, error: e as Error }
  }
}

// 使用
const result = await safeRequest<User[]>('/api/users')
if (result.success) {
  console.log(result.data)   // User[]
} else {
  console.error(result.error.message)
}
```

## 九、常见工具函数类型

```typescript
// 节流函数类型
type ThrottleFn<T extends (...args: any[]) => any> = (
  fn: T,
  delay: number
) => (...args: Parameters<T>) => void

// 防抖函数类型
type DebounceFn<T extends (...args: any[]) => any> = (
  fn: T,
  delay: number
) => (...args: Parameters<T>) => void

// Parameters：获取函数参数类型元组
type AddParams = Parameters<typeof add>  // [number, number]

// ReturnType：获取函数返回值类型
type AddReturn = ReturnType<typeof add>  // number

// 构造函数参数类型
type DateConstructorParams = ConstructorParameters<typeof Date>
```

## 十、总结

- **函数重载**：同名函数支持多种参数/返回类型组合，提供更精确的类型推断
- **this 参数**：显式声明 this 类型，避免上下文错误
- **剩余参数**：`...args: T[]` 接收不定数量参数
- **高阶函数**：函数作为参数/返回值，配合泛型保持类型安全
- **async/await**：返回 `Promise<T>`，错误处理要考虑完善
