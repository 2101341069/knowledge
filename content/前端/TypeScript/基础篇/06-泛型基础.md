# 泛型基础（Generics）

## 一、为什么需要泛型？

### 问题场景

```typescript
// ❌ 不用泛型：用 any，丢失类型
function identity(arg: any): any {
  return arg
}

const num = identity(42)      // any 类型
const str = identity('hello') // any 类型
```

```typescript
// ❌ 重复代码，不灵活
function identityNumber(arg: number): number {
  return arg
}
function identityString(arg: string): string {
  return arg
}
```

✅ **泛型解决：类型参数化**

```typescript
// ✅ 用泛型：保留类型信息
function identity<T>(arg: T): T {
  return arg
}

const num = identity(42)      // number
const str = identity('hello') // string
```

## 二、泛型基础语法

### 2.1 泛型函数

```typescript
// 完整写法
function identity<T>(arg: T): T {
  return arg
}

// 显式指定类型
let output = identity<string>('myString')

// 类型推断（推荐，自动推断）
let output = identity('myString')
```

### 2.2 多个类型参数

```typescript
function swap<T, U>(tuple: [T, U]): [U, T] {
  return [tuple[1], tuple[0]]
}

const result = swap(['hello', 10])
// result: [number, string]
```

## 三、泛型约束（Generic Constraints）

### 3.1 基础约束

```typescript
// ❌ 错误：T 不一定有 length 属性
function logLength<T>(arg: T): number {
  return arg.length  // ❌
}
```

```typescript
// ✅ 定义约束接口
interface Lengthwise {
  length: number
}

// T 必须满足 Lengthwise 接口
function logLength<T extends Lengthwise>(arg: T): number {
  return arg.length  // ✅
}

logLength('hello')   // ✅ 字符串有 length
logLength([1, 2, 3]) // ✅ 数组有 length
logLength(42)         // ❌ 数字没有
```

### 3.2  keyof 操作符约束

```typescript
// 确保 key 是对象的属性之一
function getProperty<T, K extends keyof T>(obj: T, key: K) {
  return obj[key]
}

const user = { name: 'Tom', age: 25 }
getProperty(user, 'name')  // ✅
getProperty(user, 'age')   // ✅
getProperty(user, 'email') // ❌ 'email' 不是 user 的属性
```

## 四、泛型接口

```typescript
interface Box<T> {
  value: T
  getValue(): T
  setValue(value: T): void
}

const numberBox: Box<number> = {
  value: 42,
  getValue() { return this.value },
  setValue(v) { this.value = v }
}

const stringBox: Box<string> = {
  value: 'hello',
  getValue() { return this.value },
  setValue(v) { this.value = v }
}
```

## 五、泛型类

```typescript
class Stack<T> {
  private items: T[] = []
  
  push(item: T) {
    this.items.push(item)
  }
  
  pop(): T | undefined {
    return this.items.pop()
  }
  
  peek(): T | undefined {
    return this.items[this.items.length - 1]
  }
}

// 使用
const numberStack = new Stack<number>()
numberStack.push(1)
numberStack.push(2)
numberStack.pop()  // number | undefined

const stringStack = new Stack<string>()
```

## 六、泛型默认类型

```typescript
interface ApiResponse<T = any> {
  code: number
  message: string
  data: T
}

// 使用默认类型
const res1: ApiResponse = {
  code: 200,
  message: 'ok',
  data: {}  // any 类型
}

// 指定具体类型
interface User { id: number; name: string }
const res2: ApiResponse<User> = {
  code: 200,
  message: 'ok',
  data: { id: 1, name: 'Tom' }
}
```

## 七、泛型工具类型（TypeScript 内置）

### 7.1 Partial - 所有属性可选

```typescript
interface Todo {
  title: string
  description: string
}

// 所有属性变成可选
function updateTodo(todo: Todo, fields: Partial<Todo>): Todo {
  return { ...todo, ...fields }
}

const todo = { title: 'A', description: '...' }
updateTodo(todo, { title: 'B' })  // 只传部分属性
```

### 7.2 Required - 所有属性必需

```typescript
interface Props {
  a?: number
  b?: string
}

const obj: Props = { a: 5 }  // ✅

const obj2: Required<Props> = { a: 5 }  // ❌ 缺少 b
```

### 7.3 Readonly - 所有属性只读

```typescript
interface Todo {
  title: string
}

const todo: Readonly<Todo> = {
  title: 'Hello'
}
todo.title = 'World'  // ❌ 只读
```

### 7.4 Pick - 选取部分属性

```typescript
interface User {
  id: number
  name: string
  email: string
  password: string
}

// 只选取 id 和 name
type UserPreview = Pick<User, 'id' | 'name'>
// { id: number; name: string }
```

### 7.5 Omit - 排除部分属性

```typescript
// 排除 password
type SafeUser = Omit<User, 'password'>
// { id: number; name: string; email: string }
```

### 7.6 Exclude - 从联合类型排除

```typescript
type T0 = Exclude<'a' | 'b' | 'c', 'a'>  // 'b' | 'c'
type T1 = Exclude<'a' | 'b' | 'c', 'a' | 'b'>  // 'c'
```

### 7.7 Extract - 从联合类型提取

```typescript
type T0 = Extract<'a' | 'b' | 'c', 'a' | 'f'>  // 'a'
```

### 7.8 ReturnType - 获取函数返回值类型

```typescript
function createUser() {
  return { id: 1, name: 'Tom' }
}

type User = ReturnType<typeof createUser>
// { id: number; name: string }
```
