# 表单处理

## 一、受控组件 vs 非受控组件

| | 受控组件 | 非受控组件 |
|----|----------|------------|
| 数据来源 | React state 控制 | DOM 自己管理 |
| 读取值 | 从 state 读 | 用 ref 读 DOM |
| 验证时机 | 实时验证 | 提交时验证 |
| 适用场景 | 大多数场景 | 简单表单、文件上传 |

---

## 二、受控组件（Controlled Component）

表单数据由 React state 管理，"表单元素的值 = state 的值"。

### 2.1 文本输入框

```jsx
function NameForm() {
  const [name, setName] = useState('')  // 1. 声明 state

  function handleSubmit(e) {
    e.preventDefault()
    console.log('提交的名字:', name)  // 4. 提交时从 state 读
  }

  return (
    <form onSubmit={handleSubmit}>
      <label>
        名字:
        <input
          type="text"
          value={name}          // 2. value 绑定 state
          onChange={(e) => setName(e.target.value)}  // 3. 变化时更新 state
        />
      </label>
      <button type="submit">提交</button>
    </form>
  )
}
```

### 2.2 文本域（textarea）

```jsx
function EssayForm() {
  const [essay, setEssay] = useState('请输入文章内容')

  return (
    <form>
      <label>
        文章:
        <textarea
          value={essay}
          onChange={(e) => setEssay(e.target.value)}
          rows={4}
        />
      </label>
    </form>
  )
}
```

> 💡 和原生不同，React 的 textarea 用 `value` 而不是子内容。

### 2.3 下拉选择（select）

```jsx
function FlavorForm() {
  const [value, setValue] = useState('coconut')

  const options = [
    { value: 'grapefruit', label: '葡萄柚' },
    { value: 'lime', label: '酸橙' },
    { value: 'coconut', label: '椰子' },
    { value: 'mango', label: '芒果' }
  ]

  return (
    <form>
      <label>
        选择你喜欢的口味:
        <select value={value} onChange={(e) => setValue(e.target.value)}>
          {options.map(opt => (
            <option key={opt.value} value={opt.value}>
              {opt.label}
            </option>
          ))}
        </select>
      </label>
      <p>你选择了: {value}</p>
    </form>
  )
}
```

> 💡 多选下拉：`value={[]}` + `multiple`

### 2.4 复选框（checkbox）

```jsx
function CheckboxForm() {
  const [hobbies, setHobbies] = useState(['reading'])

  function handleChange(e) {
    const value = e.target.value
    const checked = e.target.checked
    
    setHobbies(prev => 
      checked 
        ? [...prev, value]        // 选中：添加
        : prev.filter(h => h !== value)  // 取消：移除
    )
  }

  return (
    <form>
      <label>
        <input
          type="checkbox"
          value="reading"
          checked={hobbies.includes('reading')}
          onChange={handleChange}
        />
        阅读
      </label>
      <label>
        <input
          type="checkbox"
          value="coding"
          checked={hobbies.includes('coding')}
          onChange={handleChange}
        />
        编程
      </label>
      <p>已选爱好：{hobbies.join(', ')}</p>
    </form>
  )
}
```

### 2.6 处理多个输入

用一个 state 对象管理所有表单字段：

```jsx
function LoginForm() {
  const [form, setForm] = useState({
    username: '',
    password: '',
    rememberMe: false
  })

  // 通用 change 处理函数
  function handleChange(e) {
    const { name, value, type, checked } = e.target
    setForm(prev => ({
      ...prev,
      [name]: type === 'checkbox' ? checked : value
    }))
  }

  function handleSubmit(e) {
    e.preventDefault()
    console.log('提交数据:', form)
  }

  return (
    <form onSubmit={handleSubmit}>
      <input
        name="username"
        value={form.username}
        onChange={handleChange}
        placeholder="用户名"
      />
      <input
        name="password"
        type="password"
        value={form.password}
        onChange={handleChange}
        placeholder="密码"
      />
      <label>
        <input
          name="rememberMe"
          type="checkbox"
          checked={form.rememberMe}
          onChange={handleChange}
        />
        记住我
      </label>
      <button type="submit">登录</button>
    </form>
  )
}
```

## 三、非受控组件（Uncontrolled Component）

用 ref 直接从 DOM 获取值，React 不管理数据。

```jsx
import { useRef } from 'react'

function NameForm() {
  const inputRef = useRef(null)  // 1. 创建 ref

  function handleSubmit(e) {
    e.preventDefault()
    console.log('提交的名字:', inputRef.current.value)  // 3. 提交时从 DOM 读
  }

  return (
    <form onSubmit={handleSubmit}>
      <label>
        名字:
        <input
          type="text"
          ref={inputRef}  // 2. 绑定 ref 到 DOM
          defaultValue="Bob"  // 初始值用 defaultValue
        />
      </label>
      <button type="submit">提交</button>
    </form>
  )
}
```

## 四、表单验证

### 4.1 简单实时验证

```jsx
function LoginForm() {
  const [form, setForm] = useState({ username: '', password: '' })
  const [errors, setErrors] = useState({})

  // 验证函数
  function validate() {
    const newErrors = {}
    if (!form.username.trim()) {
      newErrors.username = '用户名不能为空'
    }
    if (form.password.length < 6) {
      newErrors.password = '密码至少6位'
    }
    setErrors(newErrors)
    return Object.keys(newErrors).length === 0  // 没有错误才通过
  }

  function handleSubmit(e) {
    e.preventDefault()
    if (validate()) {
      console.log('提交:', form)
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <input name="username" value={form.username} onChange={handleChange} />
        {errors.username && <span className="error">{errors.username}</span>}
      </div>
      <div>
        <input name="password" type="password" value={form.password} onChange={handleChange} />
        {errors.password && <span className="error">{errors.password}</span>}
      </div>
      <button type="submit">登录</button>
    </form>
  )
}
```

## 五、推荐的表单库

复杂表单推荐用现成的表单库，省很多事：

| 库 | 特点 |
|----|------|
| **React Hook Form** | 性能好，简单易用，非受控为主 |
| **Formik** | 生态成熟，受控为主 |
| **Zod** | 配合表单库做类型安全的验证 |

---

### React Hook Form 例子

```jsx
import { useForm } from 'react-hook-form'

function LoginForm() {
  const { register, handleSubmit, formState: { errors } } = useForm()

  function onSubmit(data) {
    console.log(data)
  }

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input
        {...register('username', { required: '用户名必填' })}
      />
      {errors.username && <span>{errors.username.message}</span>}

      <input
        type="password"
        {...register('password', { required: '密码必填', minLength: 6 })}
      />
      {errors.password && <span>{errors.password.message}</span>}

      <button type="submit">登录</button>
    </form>
  )
}
```

## 总结

✅ **受控组件**：value + onChange 绑定 state，React 管理数据（推荐）
2. ✅ **非受控组件**：ref 直接读 DOM，defaultValue 设初始值
3. ✅ 多输入字段：对象 state + 通用 handleChange
4. ✅ 实时验证：onChange 时验证并显示错误
5. ✅ 复杂表单推荐用 React Hook Form / Formik 库
