# 从零搭建企业级前端项目脚手架

## 一、为什么需要自己的脚手架

### 常见痛点

- ❌ 每次新项目都要从零配置（ESLint、TypeScript、构建工具...）
- ❌ 团队项目配置不统一，代码风格各异
- ❌ 新人上手成本高，要熟悉一大堆工具链
- ❌ 最佳实践无法沉淀，每个项目重复踩坑

### 脚手架核心价值

| 价值 | 说明 |
|------|------|
| **统一技术栈** | 团队统一技术选型（Vue/React、Vite/Webpack、TS/JS） |
| **统一代码规范** | ESLint、Prettier、Git Hooks 内置，开箱即用 |
| **最佳实践沉淀** | 目录结构、请求封装、状态管理、路由配置等内置 |
| **提升开发效率** | 一条命令创建项目，减少重复配置时间 |
| **统一构建配置** | 构建、测试、部署流程标准化 |

### 设计原则

1. ✅ **简单易用**：一条命令创建项目，交互式询问
2. ✅ **可扩展**：插件化设计，支持自定义模板
3. ✅ **可升级**：配置和模板分离，便于更新
4. ✅ **规范化**：内置代码规范和最佳实践
5. ✅ **灵活性**：支持不同框架、不同业务场景模板

          assetFileNames: 'static/[ext]/[name]-[hash].[ext]',

              }

              return 'vendor'
            }
          }
        }
      }
    }
  }

#### ② src/api/request.ts - Axios 封装

```typescript
import axios, { AxiosRequestConfig, AxiosResponse, AxiosError } from 'axios'
import { ElMessage, ElMessageBox } from 'element-plus'
import { useUserStore } from '@stores/user'

// 创建实例
const service = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 30000
})

// 请求拦截器
service.interceptors.request.use(
  (config: AxiosRequestConfig) => {
    const userStore = useUserStore()
    
    // 添加 Token
    if (userStore.token) {
      config.headers!.Authorization = `Bearer ${userStore.token}`
    }
    
    return config
  },
  (error: AxiosError) => Promise.reject(error)
)

// 响应拦截器
service.interceptors.response.use(
  (response: AxiosResponse) => {
    const { code, data, message } = response.data
    
    // 请求成功
    if (code === 0 || code === 200) {
      return data
    }
    
    // 业务错误
    ElMessage.error(message || '请求失败')
    return Promise.reject(new Error(message || '请求失败'))
  },
  (error: AxiosError) => {
    const status = error.response?.status
    const userStore = useUserStore()
    
    switch (status) {
      case 401:
        // Token 过期，跳转到登录
        ElMessageBox.confirm(
          '登录状态已过期，请重新登录',
          '系统提示',
          { confirmButtonText: '重新登录', showCancelButton: false, type: 'warning' }
        ).then(() => {
          userStore.logout()
          location.reload()
        })
        break
        
      case 403:
        ElMessage.error('没有权限访问该资源')
        break
        
      case 404:
        ElMessage.error('请求的资源不存在')
        break
        
      case 500:
        ElMessage.error('服务器错误')
        break
        
      default:
        ElMessage.error(error.message || '网络错误')
    }
    
    return Promise.reject(error)
  }
)

// 导出请求方法
export default {
  get<T = any>(url: string, params?: any, config?: AxiosRequestConfig): Promise<T> {
    return service.get(url, { params, ...config })
  },
  
  post<T = any>(url: string, data?: any, config?: AxiosRequestConfig): Promise<T> {
    return service.post(url, data, config)
  },
  
  put<T = any>(url: string, data?: any, config?: AxiosRequestConfig): Promise<T> {
    return service.put(url, data, config)
  },
  
  delete<T = any>(url: string, params?: any, config?: AxiosRequestConfig): Promise<T> {
    return service.delete(url, { params, ...config })
  }
}
```

#### ③ src/router/index.ts - 路由配置

```typescript
import { createRouter, createWebHistory, RouteRecordRaw } from 'vue-router'
import { useUserStore } from '@stores/user'

// 基础路由
const baseRoutes: RouteRecordRaw[] = [
  {
    path: '/login',
    name: 'Login',
    component: () => import('@/views/Login/index.vue'),
    meta: { requiresAuth: false, title: '登录' }
  },
  {
    path: '/',
    name: 'Layout',
    component: () => import('@/layout/index.vue'),
    redirect: '/dashboard',
    children: [
      {
        path: 'dashboard',
        name: 'Dashboard',
        component: () => import('@/views/Dashboard/index.vue'),
        meta: { title: '仪表盘', icon: 'dashboard' }
      }
    ]
  },
  {
    path: '/404',
    name: '404',
    component: () => import('@/views/404.vue'),
    meta: { title: '页面不存在' }
  },
  // 404 兜底路由
  {
    path: '/:pathMatch(.*)*',
    redirect: '/404'
  }
]

const router = createRouter({
  history: createWebHistory(import.meta.env.VITE_BASE_URL),
  routes: baseRoutes,
  // 滚动行为
  scrollBehavior(to, from, savedPosition) {
    if (savedPosition) {
      return savedPosition
    }
    return { top: 0 }
  }
})

// 路由守卫
router.beforeEach((to, from, next) => {
  const userStore = useUserStore()
  
  // 设置页面标题
  document.title = `${to.meta.title || '管理系统'} - ${import.meta.env.VITE_APP_TITLE}`
  
  // 不需要登录的页面直接放行
  if (to.meta.requiresAuth === false) {
    next()
    return
  }
  
  // 未登录
  if (!userStore.token) {
    next({ name: 'Login', query: { redirect: to.fullPath } })
    return
  }
  
  // 已登录访问登录页，跳转到首页
  if (to.name === 'Login') {
    next({ name: 'Dashboard' })
    return
  }
  
  // 权限校验（根据实际业务实现）
  if (to.meta.roles && !hasAnyRole(to.meta.roles as string[])) {
    next({ name: '403' })
    return
  }
  
  next()
})

function hasAnyRole(roles: string[]): boolean {
  const userStore = useUserStore()
  return roles.some(role => userStore.roles?.includes(role))
}

export default router
```

#### ④ src/stores/user.ts - 用户状态管理

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { login, getUserInfo } from '@api/modules/user'

export const useUserStore = defineStore('user', () => {
  // ===== State =====
  const token = ref<string>(localStorage.getItem('token') || '')
  const userInfo = ref<UserInfo | null>(null)
  const roles = ref<string[]>([])
  const permissions = ref<string[]>([])

  // ===== Getters =====
  const isLoggedIn = computed(() => !!token.value)
  const userName = computed(() => userInfo.value?.name || '')
  const avatar = computed(() => userInfo.value?.avatar || '')

  // ===== Actions =====
  // 登录
  async function doLogin(credentials: LoginParams) {
    const result = await login(credentials)
    token.value = result.token
    localStorage.setItem('token', result.token)
    // 获取用户信息
    await fetchUserInfo()
    return result
  }

  // 获取用户信息
  async function fetchUserInfo() {
    const info = await getUserInfo()
    userInfo.value = info
    roles.value = info.roles || []
    permissions.value = info.permissions || []
    return info
  }

  // 退出登录
  function logout() {
    token.value = ''
    userInfo.value = null
    roles.value = []
    permissions.value = []
    localStorage.removeItem('token')
  }

  return {
    // State
    token,
    userInfo,
    roles,
    permissions,
    // Getters
    isLoggedIn,
    userName,
    avatar,
    // Actions
    doLogin,
    fetchUserInfo,
    logout
  }
})
```

#### ⑤ composables/useTable.ts - 表格组合式函数

```typescript
import { ref, reactive, computed } from 'vue'

interface UseTableOptions<T> {
  // 请求函数
  fetchFn: (params: any) => Promise<{ list: T[]; total: number }>
  // 初始参数
  initialParams?: Record<string, any>
  // 是否立即请求
  immediate?: boolean
}

export function useTable<T = any>(options: UseTableOptions<T>) {
  const { fetchFn, initialParams = {}, immediate = true } = options

  // ===== State =====
  const loading = ref(false)
  const dataSource = ref<T[]>([])
  const total = ref(0)

  // 分页
  const pagination = reactive({
    page: 1,
    pageSize: 10
  })

  // 查询参数
  const searchParams = ref<Record<string, any>>({ ...initialParams })

  // ===== Getters =====
  const hasData = computed(() => dataSource.value.length > 0)

  // ===== Actions =====
  async function fetchData() {
    loading.value = true
    try {
      const params = {
        page: pagination.page,
        pageSize: pagination.pageSize,
        ...searchParams.value
      }
      const result = await fetchFn(params)
      dataSource.value = result.list
      total.value = result.total
    } finally {
      loading.value = false
    }
  }

  // 搜索
  function handleSearch() {
    pagination.page = 1
    fetchData()
  }

  // 重置搜索
  function handleReset() {
    Object.assign(searchParams.value, initialParams)
    handleSearch()
  }

  // 分页变化
  function handlePageChange(page: number) {
    pagination.page = page
    fetchData()
  }

  function handlePageSizeChange(pageSize: number) {
    pagination.pageSize = pageSize
    pagination.page = 1
    fetchData()
  }

  // 刷新
  function refresh() {
    fetchData()
  }

  // 立即执行
  if (immediate) {
    fetchData()
  }

  return {
    // State
    loading,
    dataSource,
    total,
    pagination,
    searchParams,
    // Getters
    hasData,
    // Actions
    fetchData,
    handleSearch,
    handleReset,
    handlePageChange,
    handlePageSizeChange,
    refresh
  }
}
```

使用示例：
```vue
<script setup lang="ts">

const {
  loading,
  dataSource,
  pagination,
  searchParams,
  handleSearch,
  handleReset,

  handlePageSizeChange

})

// src/commands/create.ts

interface CreateOptions {

  force?: boolean

interface PromptAnswers {
  projectName: string
  template: string
  description: string

// 模板列表

          name: 'packageManager',
          message: '请选择包管理器:',
          choices: ['pnpm', 'npm', 'yarn'],
          default: 'pnpm'
        }
      ])

      // 2. 检查目录是否存在
      const targetDir = path.resolve(process.cwd(), projectName)
      if (fs.existsSync(targetDir)) {
        if (options.force) {
          fs.rmSync(targetDir, { recursive: true, force: true })
        } else {
          const { overwrite } = await inquirer.prompt([
            {
              type: 'confirm',
              name: 'overwrite',
              message: `目录 ${projectName} 已存在，是否覆盖？`,
              default: false
            }
          ])
          
          if (!overwrite) {
            console.log(chalk.yellow('\n❌  操作取消\n'))
            return
          }
          
          fs.rmSync(targetDir, { recursive: true, force: true })
        }
      }

      // 3. 下载模板
      const templateName = options.template || answers.template
      const spinner = ora(`下载模板 ${templateName}...`).start()

      // 从 GitHub 下载模板（这里用本地文件或远程仓库）
      const templateRepo = `github:username/repo/templates/${templateName}#main`

      download(templateRepo, targetDir, { clone: false }, async (err: any) => {
        if (err) {
          spinner.fail('模板下载失败')
          console.error(err)
          return
        }

        spinner.succeed('模板下载完成')

        // 4. 模板渲染（替换 package.json 中的变量）
        const packageJsonPath = path.join(targetDir, 'package.json')
        const packageJsonContent = fs.readFileSync(packageJsonPath, 'utf-8')
        const template = handlebars.compile(packageJsonContent)
        const result = template({
          name: projectName,
          description: answers.description,
          author: answers.author
        })
        fs.writeFileSync(packageJsonPath, result)

        console.log(`\n✅  项目 ${chalk.green(projectName)} 创建成功!`)
        console.log(`\n📝  接下来执行:\n`)
        console.log(`  cd ${projectName}`)
        console.log(`  ${answers.packageManager} install`)
        console.log(`  ${answers.packageManager} run dev\n`)
      })
    })
}
```
